local Workspace = game:GetService("Workspace")

local HttpService = game:GetService("HttpService")

    AntiRagdoll = true,
    SpeedBypass = false,
    SpeedValue = 60,
    AntiTrap = true,
    FixLag = true,
    Noclip = true,
    ShowPetInEgg = true,
    AutoGetEgg = false,
    SelectedEggFilter = "Secret",

    AutoAttack = false,

}

task.spawn(function()
    while task.wait(1) do
        if getgenv().KhangNguyenConfig.AntiRagdoll then
            pcall(function()
                local char = LocalPlayer.Character
                if char then
                    local humanoid = char:FindFirstChildOfClass("Humanoid")
                    if humanoid then
                        humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                        humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                        humanoid:SetStateEnabled(Enum.HumanoidStateType.PlatformStand, false)
                        humanoid:SetStateEnabled(Enum.HumanoidStateType.Stunned, false)
                    end
                end
            end)
        end
    end
end)

RunService.RenderStepped:Connect(function(dt)
    if getgenv().KhangNguyenConfig.SpeedBypass then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                if hrp and humanoid then
                    local moveDir = humanoid.MoveDirection
                    if moveDir.Magnitude > 0 then
                        hrp.CFrame = hrp.CFrame + (moveDir * getgenv().KhangNguyenConfig.SpeedValue * dt)
                    end
                end
            end
        end)
    end
end)

RunService.Stepped:Connect(function()
    if getgenv().KhangNguyenConfig.Noclip then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
        end)
    end
end)

task.spawn(function()
    while task.wait(0.8) do
        if getgenv().KhangNguyenConfig.AntiTrap then
            pcall(function()
                for _, v in pairs(Workspace:GetDescendants()) do
                    if v and v.Parent and v:IsA("BasePart") then
                        local name = v.Name:lower()
                        if name:find("trap") or name:find("bẫy") or name:find("spike") or name:find("danger") then
                            v.CanCollide = false
                            v.Transparency = 1
                            local touch = v:FindFirstChild("TouchInterest")
                            if touch then touch:Destroy() end
                        end
                    end
                end
            end)
        end
    end
end)

local function ApplyKhangNguyenFixLag()
    pcall(function()
        local terrain = Workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            terrain.WaterWaveSize = 0
            terrain.WaterWaveTransparency = 1
            terrain.WaterReflectance = 0
        end
        pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
        for _, v in pairs(Workspace:GetDescendants()) do
            if v and v.Parent then
                if v:IsA("BasePart") then
                    v.Material = Enum.Material.SmoothPlastic
                    v.Reflectance = 0
                    local name = v.Name:lower()
                    if name:find("tree") or name:find("rock") or name:find("stone") or name:find("cay") or name:find("da") or name:find("cloud") then
                        v:Destroy()
                    end
                elseif v:IsA("Decal") or v:IsA("Texture") then
                    v.Transparency = 1
                elseif v:IsA("ParticleEmitter") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then
                    v.Enabled = false
                end
            end
        end
    end)
end
if getgenv().KhangNguyenConfig.FixLag then task.spawn(ApplyKhangNguyenFixLag) end

local EggCache = {}
local function ScanAndManageEggs()
    EggCache = {}
    pcall(function()
        for _, egg in pairs(Workspace:GetDescendants()) do
            if egg and egg.Parent and egg:IsA("Model") then
                local nameLower = egg.Name:lower()
                if nameLower:find("egg") or nameLower:find("trứng") then
                    local primaryPart = egg.PrimaryPart or egg:FindFirstChildWhichIsA("BasePart")
                    if primaryPart then
                        local data = {
                            Instance = egg,
                            Name = egg.Name,
                            Position = primaryPart.Position,
                            IsSecret = nameLower:find("secret") ~= nil,
                            IsMutation = nameLower:find("mutation") ~= nil or nameLower:find("đột biến") ~= nil,
                            IsEvent = nameLower:find("event") ~= nil or nameLower:find("hoa nở") ~= nil,
                            IsParasite = nameLower:find("parasite") ~= nil or nameLower:find("ký sinh") ~= nil,
                            IsMonster = nameLower:find("monster") ~= nil or nameLower:find("quái vật") ~= nil,
                        }
                        table.insert(EggCache, data)

                        if not egg:FindFirstChild("KhangNguyenEggESP") then
                            local highlight = Instance.new("Highlight")
                            highlight.Name = "KhangNguyenEggESP"
                            highlight.Adornee = egg
                            highlight.FillColor = data.IsSecret and Color3.fromRGB(255, 0, 0) or (data.IsMutation and Color3.fromRGB(0, 255, 255) or Color3.fromRGB(255, 215, 0))
                            highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                            highlight.Parent = egg

                            local billboard = Instance.new("BillboardGui")
                            billboard.Name = "KhangNguyenTagUI"
                            billboard.Size = UDim2.new(0, 140, 0, 70)
                            billboard.AlwaysOnTop = true
                            billboard.Adornee = primaryPart

                            local label = Instance.new("TextLabel")
                            label.Size = UDim2.new(1, 0, 1, 0)
                            label.BackgroundTransparency = 1
                            label.TextColor3 = Color3.fromRGB(255, 255, 255)
                            label.TextScaled = true
                            label.TextStrokeTransparency = 0
                            label.Text = egg.Name .. (getgenv().KhangNguyenConfig.ShowPetInEgg and "\n[Khang Nguyen Pet]" or "")
                            label.Parent = billboard

                            billboard.Parent = egg
                        end
                    end
                end
            end
        end
    end)
end

local function SmoothTeleport(targetCF)
    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        
        local distance = (hrp.Position - targetCF.Position).Magnitude
        local speed = math.clamp(getgenv().KhangNguyenConfig.FlySpeed, 50, 1000)
        local timeTaken = math.clamp(distance / speed, 0.05, 2)
        local startTime = tick()
        local startCF = hrp.CFrame
        
        while tick() - startTime < timeTaken do
            if not char or not hrp.Parent then break end
            local alpha = (tick() - startTime) / timeTaken
            hrp.CFrame = startCF:Lerp(targetCF, alpha)
            RunService.RenderStepped:Wait()
        end
        if hrp and hrp.Parent then hrp.CFrame = targetCF end
    end)
end

task.spawn(function()
    while task.wait(0.5) do
        if getgenv().KhangNguyenConfig.AutoGetEgg then
            pcall(function()
                ScanAndManageEggs()
                local bestEgg = nil
                for _, egg in ipairs(EggCache) do
                    if egg.IsSecret then bestEgg = egg; break; end
                end
                if not bestEgg then
                    for _, egg in ipairs(EggCache) do
                        local filter = getgenv().KhangNguyenConfig.SelectedEggFilter
                        if filter == "Mutation" and egg.IsMutation then bestEgg = egg; break;
                        elseif filter == "Event" and egg.IsEvent then bestEgg = egg; break;
                        elseif filter == "Parasite" and egg.IsParasite then bestEgg = egg; break;
                        elseif filter == "Monster" and egg.IsMonster then bestEgg = egg; break;
                        end
                    end
                end
                if not bestEgg and #EggCache > 0 then
                    bestEgg = EggCache[math.random(1, #EggCache)]
                end
                if bestEgg and bestEgg.Instance and bestEgg.Instance.Parent then
                    local char = LocalPlayer.Character
                    if char then
                        local hrp = char:FindFirstChild("HumanoidRootPart")
                        if hrp then
                            local homePosition = hrp.CFrame
                            SmoothTeleport(CFrame.new(bestEgg.Position + Vector3.new(0, 4, 0)))
                            task.wait(0.2)
                            local tool = char:FindFirstChildOfClass("Tool")
                            if tool then pcall(function() tool:Activate() end) end
                            task.wait(0.25)
                            SmoothTeleport(homePosition)
                        end
                    end
                end
            end)
        end
    end
end)

        if getgenv().KhangNguyenConfig.AutoServerHop then
            pcall(function()

                local success, result = pcall(function() return game:HttpGet(api) end)
                if success and result then
                    local decodeSuccess, response = pcall(function() return HttpService:JSONDecode(result) end)
                    if decodeSuccess and response and response.data then
                        for _, s in ipairs(response.data) do
                            if s.playing and s.maxPlayers and s.playing < s.maxPlayers and s.playing <= getgenv().KhangNguyenConfig.MinPlayersHop and s.id ~= game.JobId then
                                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                                break
                            end
                        end
                    end
                end
            end)
        end
    end
end)
