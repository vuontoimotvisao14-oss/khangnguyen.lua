-- [MODULE 1]: Metatable Hooking Chống Check Thuộc Tính
local OriginalIndex
OriginalIndex = hookmetamethod(game, "__index", function(self, property)
    if not checkcaller() and (property == "WalkSpeed" or property == "JumpPower" or property == "HipHeight") then
        local char = LocalPlayer.Character
        if char and self == char:FindFirstChildOfClass("Humanoid") then
            if property == "WalkSpeed" then return 16 end
            if property == "JumpPower" then return 50 end
        end
    end
    return OriginalIndex(self, property)
end)

-- [MODULE 2]: Dynamic Memory Pool Dọn Rác Tự Động
local MemoryPool = {
    ActiveInstances = {},
    Register = function(self, instance) table.insert(self.ActiveInstances, instance) end,
    Flush = function(self)
        for i = #self.ActiveInstances, 1, -1 do
            if not self.ActiveInstances[i] or not self.ActiveInstances[i].Parent then
                table.remove(self.ActiveInstances, i)
            end
        end
        collectgarbage("step", 10)
    end
}
task.spawn(function() while task.wait(5) do MemoryPool:Flush() end end)

local function InitializeMasterFixLag()
    pcall(function()
        local terrain = Workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            terrain.WaterWaveSize = 0
            terrain.WaterWaveTransparency = 1
            terrain.WaterReflectance = 0
            terrain.WaterColor = Color3.new(0, 0, 0)
        end
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 999999
        Lighting.Brightness = 4
        Lighting.ClockTime = 12
        
        for _, worldObject in pairs(Workspace:GetDescendants()) do
            if worldObject and worldObject.Parent then
                if worldObject:IsA("BasePart") then
                    worldObject.Material = Enum.Material.SmoothPlastic
                    worldObject.Reflectance = 0
                    worldObject.CastShadow = false
                elseif worldObject:IsA("Decal") or worldObject:IsA("Texture") then
                    worldObject.Transparency = 1
                elseif worldObject:IsA("ParticleEmitter") or worldObject:IsA("Fire") or worldObject:IsA("Smoke") or worldObject:IsA("Sparkles") or worldObject:IsA("Trail") then
                    worldObject.Enabled = false
                end
            end
        end
    end)
end

-- [MODULE 7]: TweenService Hyper Fly Siêu Mượt Mà
local function ExecuteTweenFly(targetCFrame)
    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        local dist = (root.Position - targetCFrame.Position).Magnitude
        local speed = 3000
        local timeDuration = math.clamp(dist / speed, 0.01, 0.12)
        
        local tweenInfo = TweenInfo.new(timeDuration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
        local tween = TweenService:Create(root, tweenInfo, {CFrame = targetCFrame})
        tween:Play()
        tween.Completed:Wait()
    end)
end

    while task.wait(0.05) do
        pcall(function()
            if getgenv().KhangNguyenSettings.TrollMode.Enabled and getgenv().KhangNguyenSettings.TrollMode.TargetPlayerName ~= "" then
                return
            end
            
            local bestTargetPart = nil
            local bestTargetModel = nil
            local highestScore = -math.huge
            
            for _, obj in pairs(Workspace:GetDescendants()) do
                if obj:IsA("Model") and obj.Parent then
                    local nameLower = obj.Name:lower()
                    if nameLower:find("egg") or nameLower:find("trứng") then
                        local primary = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                        if primary then
                            local matchesRarity = false
                            local matchesMutation = false
                            local computedScore = 0
                            
                            for rarityName, isEnabled in pairs(getgenv().KhangNguyenSettings.AutoFarmRarity) do
                                if isEnabled and nameLower:find(rarityName:lower()) then
                                    matchesRarity = true
                                    computedScore = computedScore + 500
                                    break
                                end
                            end
                            
                            for mutationName, isEnabled in pairs(getgenv().KhangNguyenSettings.AutoFarmMutation) do
                                if isEnabled and nameLower:find(mutationName:lower()) then
                                    matchesMutation = true
                                    computedScore = computedScore + 1000
                                    break
                                end
                            end
                            
                            if matchesRarity or matchesMutation or (not next(getgenv().KhangNguyenSettings.AutoFarmRarity) and not next(getgenv().KhangNguyenSettings.AutoFarmMutation)) then
                                local char = LocalPlayer.Character
                                if char and char:FindFirstChild("HumanoidRootPart") then
                                    local dist = (char.HumanoidRootPart.Position - primary.Position).Magnitude
                                    computedScore = computedScore - (dist * 0.05)
                                    
                                    if computedScore > highestScore then
                                        highestScore = computedScore
                                        bestTargetPart = primary
                                        bestTargetModel = obj
                                    end
                                end
                            end
                        end
                    end
                end
            end
            
            if bestTargetPart and bestTargetModel then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local originalCF = char.HumanoidRootPart.CFrame
                    
                    ExecuteTweenFly(CFrame.new(bestTargetPart.Position + Vector3.new(0, 3, 0)))
                    task.wait(0.01)
                    
                    for _, prompt in pairs(bestTargetModel:GetDescendants()) do
                        if prompt:IsA("ProximityPrompt") then
                            fireproximityprompt(prompt)
                        end
                    end
                    
                    local tool = char:FindFirstChildOfClass("Tool") or LocalPlayer.Backpack:FindFirstChildOfClass("Tool")
                    if tool then
                        if tool.Parent == LocalPlayer.Backpack then tool.Parent = char end
                        pcall(function() tool:Activate() end)
                    end
                    
                    task.wait(0.01)
                    ExecuteTweenFly(originalCF)
                end
            end
        end)
    end
end)
