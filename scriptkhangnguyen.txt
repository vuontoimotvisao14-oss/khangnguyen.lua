local function CreateUIToggle(name, defaultState, callback)
    local ToggleBtn = Instance.new("TextButton")
    ToggleBtn.Size = UDim2.new(1, 0, 0, 40)
    ToggleBtn.BackgroundColor3 = defaultState and Color3.fromRGB(0, 120, 180) or Color3.fromRGB(40, 40, 50)
    ToggleBtn.BorderSizePixel = 0
    ToggleBtn.Text = name .. ": " .. (defaultState and "[BẬT]" or "[TẮT]")
    ToggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    ToggleBtn.TextSize = 13
    ToggleBtn.Font = Enum.Font.GothamSemibold
    ToggleBtn.Parent = ScrollingFrame
    
    local BtnCorner = Instance.new("UICorner")
    BtnCorner.CornerRadius = UDim.new(0, 8)
    BtnCorner.Parent = ToggleBtn
    
    local state = defaultState
    ToggleBtn.MouseButton1Click:Connect(function()
        state = not state
        ToggleBtn.BackgroundColor3 = state and Color3.fromRGB(0, 120, 180) or Color3.fromRGB(40, 40, 50)
        ToggleBtn.Text = name .. ": " .. (state and "[BẬT]" or "[TẮT]")
        callback(state)
    end)
end

local OriginalIndex
OriginalIndex = hookmetamethod(game, "__index", function(self, property)
    if not checkcaller() and (property == "WalkSpeed" or property == "JumpPower" or property == "HipHeight") then
        local char = LocalPlayer.Character
        if char and self == char:FindFirstChildOfClass("Humanoid") then
            if property == "WalkSpeed" then return 16 end
            if property == "JumpPower" then return 50 end
        end
    end
    return OriginalIndex(self, property)
end)

-- [MODULE 2]: Dynamic Memory Pool Dọn Rác Tự Động
local MemoryPool = {
    ActiveInstances = {},
    Register = function(self, instance) table.insert(self.ActiveInstances, instance) end,
    Flush = function(self)
        for i = #self.ActiveInstances, 1, -1 do
            if not self.ActiveInstances[i] or not self.ActiveInstances[i].Parent then
                table.remove(self.ActiveInstances, i)
            end
        end
        collectgarbage("step", 10)
    end
}
task.spawn(function() while task.wait(5) do MemoryPool:Flush() end end)

-- [MODULE 6]: Master Fix Lag & Tối Ưu Hóa Hiệu Năng
local function InitializeMasterFixLag()
    pcall(function()
        local terrain = Workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            terrain.WaterWaveSize = 0
            terrain.WaterWaveTransparency = 1
            terrain.WaterReflectance = 0
            terrain.WaterColor = Color3.new(0, 0, 0)
        end
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 999999
        Lighting.Brightness = 4
        Lighting.ClockTime = 12
        
        for _, worldObject in pairs(Workspace:GetDescendants()) do
            if worldObject and worldObject.Parent then
                if worldObject:IsA("BasePart") then
                    worldObject.Material = Enum.Material.SmoothPlastic
                    worldObject.Reflectance = 0
                    worldObject.CastShadow = false
                elseif worldObject:IsA("Decal") or worldObject:IsA("Texture") then
                    worldObject.Transparency = 1
                elseif worldObject:IsA("ParticleEmitter") or worldObject:IsA("Fire") or worldObject:IsA("Smoke") or worldObject:IsA("Sparkles") or worldObject:IsA("Trail") then
                    worldObject.Enabled = false
                end
            end
        end
    end)
end
task.spawn(InitializeMasterFixLag)

-- [MODULE 7]: TweenService Hyper Fly Siêu Mượt Mà
local function ExecuteTweenFly(targetCFrame)
    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        local dist = (root.Position - targetCFrame.Position).Magnitude
        local speed = 3000
        local timeDuration = math.clamp(dist / speed, 0.01, 0.12)
        
        local tweenInfo = TweenInfo.new(timeDuration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
        local tween = TweenService:Create(root, tweenInfo, {CFrame = targetCFrame})
        tween:Play()
        tween.Completed:Wait()
    end)
end
