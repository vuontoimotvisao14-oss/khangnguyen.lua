local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local TeleportService = game:GetService("TeleportService")
local HttpService = game:GetService("HttpService")
local LocalPlayer = Players.LocalPlayer

print("==================================================")
print("[Khang Nguyen] ĐÃ KÍCH HOẠT TẤT CẢ TÍNH NĂNG NGẦM!")
print("==================================================")

-- 1. Tự động bật Noclip (Đi xuyên tường)
RunService.Stepped:Connect(function()
    pcall(function()
        local char = LocalPlayer.Character
        if char then
            for _, part in pairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
        end
    end)
end)

-- 2. Tự động bật Anti-Ragdoll & Chống ngã
task.spawn(function()
    while task.wait(1) do
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.PlatformStand, false)
                end
            end
        end)
    end
end)

-- 3. Tự động vô hiệu hóa bẫy (Anti-Trap)
task.spawn(function()
    while task.wait(0.8) do
        pcall(function()
            for _, v in pairs(Workspace:GetDescendants()) do
                if v and v.Parent and v:IsA("BasePart") then
                    local name = v.Name:lower()
                    if name:find("trap") or name:find("bẫy") or name:find("spike") or name:find("danger") then
                        v.CanCollide = false
                        v.Transparency = 1
                        local touch = v:FindFirstChild("TouchInterest")
                        if touch then touch:Destroy() end
                    end
                end
            end
        end)
    end
end)

-- 4. Tự động tối ưu giảm lag (Fix Lag)
pcall(function()
    for _, v in pairs(Workspace:GetDescendants()) do
        if v and v.Parent then
            if v:IsA("BasePart") then
                v.Material = Enum.Material.SmoothPlastic
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v.Transparency = 1
            elseif v:IsA("ParticleEmitter") or v:IsA("Fire") or v:IsA("Smoke") then
                v.Enabled = false
            end
        end
    end
end)

-- 5. Hàm bay mượt (Smooth Fly)
local function ForceFlyTo(targetCF)
    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        
        local distance = (hrp.Position - targetCF.Position).Magnitude
        local speed = 450
        local timeTaken = math.clamp(distance / speed, 0.05, 1.2)
        local startTime = tick()
        local startCF = hrp.CFrame
        
        while tick() - startTime < timeTaken do
            if not char or not hrp.Parent then break end
            local alpha = (tick() - startTime) / timeTaken
            hrp.CFrame = startCF:Lerp(targetCF, alpha)
            RunService.RenderStepped:Wait()
        end
        if hrp and hrp.Parent then hrp.CFrame = targetCF end
    end)
end

-- 6. Tự động quét và Auto Farm Trứng (Ưu tiên Secret/Cuối map)
task.spawn(function()
    while task.wait(0.6) do
        pcall(function()
            local bestEggPart = nil
            for _, obj in pairs(Workspace:GetDescendants()) do
                if obj and obj.Parent and obj:IsA("Model") then
                    local nameLower = obj.Name:lower()
                    if nameLower:find("egg") or nameLower:find("trứng") then
                        local primaryPart = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                        if primaryPart then
                            if nameLower:find("secret") or nameLower:find("mutation") or nameLower:find("final") or nameLower:find("cuối") then
                                bestEggPart = primaryPart
                                break
                            elseif not bestEggPart then
                                bestEggPart = primaryPart
                            end
                        end
                    end
                end
            end

            if bestEggPart then
                local char = LocalPlayer.Character
                if char then
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        local oldPos = hrp.CFrame
                        ForceFlyTo(CFrame.new(bestEggPart.Position + Vector3.new(0, 4, 0)))
                        task.wait(0.2)
                        local tool = char:FindFirstChildOfClass("Tool")
                        if tool then pcall(function() tool:Activate() end) end
                        task.wait(0.2)
                        ForceFlyTo(oldPos)
                    end
                end
            end
        end)
    end
end)

-- 7. Mở rộng Hitbox người chơi tự động
RunService.RenderStepped:Connect(function()
    pcall(function()
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    hrp.Size = Vector3.new(25, 25, 25)
                    hrp.Transparency = 0.8
                    hrp.CanCollide = false
                end
            end
        end
    end)
end)
